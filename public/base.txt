const btnToggle = document.querySelector('.navbar__theme-toggle')
const menu = document.querySelector('.navbar__menu')
const close = document.querySelector('.navbar__close')
const themeIcon = document.querySelector('.navbar__icon--toggle')

menu.addEventListener('click', openMenu)
close.addEventListener('click', closeMenu)
btnToggle.addEventListener('click', themeToggle)

function openMenu(){
    document.querySelector('.navbar__list').classList.add('navbar__list--active')
}
function closeMenu(){
     document.querySelector('.navbar__list').classList.remove('navbar__list--active')
}

function applyTheme(isDark){
    const body = document.body
    if(isDark){
        body.classList.add('theme--dark')
        themeIcon.textContent = 'wb_sunny'
    }else{
        body.classList.remove('theme--dark')
        themeIcon.textContent = 'moon_stars'
    }
}

function themeToggle(){
    const body = document.body;
    const isDark = !body.classList.contains('theme--dark')

    applyTheme(isDark)
    localStorage.setItem('theme', isDark ? 'dark': 'light')
}

function initTheme(){
    const savedTheme = localStorage.getItem('theme')
    const prefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches
    
    const shouldBeDark = savedTheme ? savedTheme === 'dark': prefersDark
    
    applyTheme(shouldBeDark)
}

document.addEventListener('DOMContentLoaded', initTheme)
